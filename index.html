// --- BULLETPROOF STATE INITIALIZER & SANITIZER --- function loadLocalState() { try { const local = localStorage.getItem('nsl_s2_verified_state_v4'); if (local) { const parsed = JSON.parse(local); // Ensure teams and player arrays are valid if (parsed.teams && typeof parsed.teams === 'object') { appState.teams = { ...OFFICIAL_TEAMS, ...parsed.teams }; Object.keys(appState.teams).forEach(k => { if (!Array.isArray(appState.teams[k].players)) { appState.teams[k].players = OFFICIAL_TEAMS[k]?.players || []; } }); } else { appState.teams = OFFICIAL_TEAMS; } // Ensure fixtures is a valid non-empty array if (Array.isArray(parsed.fixtures) && parsed.fixtures.length > 0) { appState.fixtures = parsed.fixtures; } else { appState.fixtures = OFFICIAL_FIXTURES; } appState.rules = parsed.rules || OFFICIAL_RULES; appState.activeMatch = parsed.activeMatch || null; } else { appState.teams = OFFICIAL_TEAMS; appState.fixtures = OFFICIAL_FIXTURES; appState.rules = OFFICIAL_RULES; } appState.isAdmin = localStorage.getItem('nsl_s2_admin_logged') === 'true'; } catch (e) { console.error('State parse recovery triggered:', e); appState.teams = OFFICIAL_TEAMS; appState.fixtures = OFFICIAL_FIXTURES; appState.rules = OFFICIAL_RULES; } }
